# 02_bulk_wgcna_immune.R
# WGCNA on TCGA-LIHC with an IMMUNE trait (paper Fig 1)
#
# Key change: modules are chosen by their correlation with an immune infiltration
# score (MCP-counter) measured WITHIN TUMOR samples, not by Tumor vs Normal.
# (Tumor vs Normal modules are proliferation modules, which is how CDKN3/MND1
# ended up as "hub genes".)
#
# The network itself (blockwiseModules) is expensive. If
# rds/LIHC_WGCNA_network.RData already exists it is reused (set
# rebuild_network <- TRUE to recompute it from scratch).
# Output: rds/LIHC_WGCNA_results.RData, results/HCC_coexpressed_genes.csv
# =============================================================================
.libPaths(c("D:/Bioinfo/R/library", .libPaths()))
source("D:/Bioinfo/scripts/config.R")
library(WGCNA)
library(dplyr)
library(MCPcounter)
options(stringsAsFactors = FALSE)
allowWGCNAThreads(nThreads = 4)

# ---- settings -----------------------------------------------------------------
rebuild_network <- FALSE
imm_cor_cutoff  <- 0.4    # min module correlation with ImmuneScore (tumor samples only)
n_modules       <- 2      # how many immune modules to keep (strongest first)
gs_cut          <- 0.3    # keep module genes with |GS for ImmuneScore| >= gs_cut
n_heat          <- 400    # genes in the network heatmap

expr_data <- readRDS(P("rds", "TCGA_LIHC_expr.rds"))          # genes x samples (log2 RSEM+1)

# ---- 1. immune scores for every sample (MCP-counter) -------------------------------
mcp   <- MCPcounter.estimate(as.matrix(expr_data), featuresType = "HUGO_symbols")
mcp_t <- as.data.frame(t(mcp), check.names = FALSE)           # samples x cell populations
immune_cols <- colnames(mcp_t)[!grepl("endothelial|fibroblast", colnames(mcp_t), ignore.case = TRUE)]
cat("Immune populations in the composite score:", paste(immune_cols, collapse = ", "), "\n")
immune_z <- rowMeans(scale(mcp_t[, immune_cols, drop = FALSE]))   # composite ImmuneScore
write.csv(data.frame(Sample = rownames(mcp_t), ImmuneScore = immune_z, mcp_t, check.names = FALSE),
          P("results", "immune_scores_MCPcounter.csv"), row.names = FALSE)

# ---- 2. network: reuse or build ------------------------------------------------------
net_file <- P("rds", "LIHC_WGCNA_network.RData")
if (file.exists(net_file) && !rebuild_network) {
  load(net_file)       # net, expr_clean, traitData, mergedColors, soft_power (clinical_matched if saved)
  cat("Loaded existing network:", nrow(expr_clean), "samples x", ncol(expr_clean), "genes\n")
} else {
  expr_t <- t(expr_data)                                      # samples x genes
  expr_t <- expr_t[, colMeans(expr_t > 0) >= 0.5]             # genes expressed in >= 50% of samples
  gsg <- goodSamplesGenes(expr_t, verbose = 3)
  if (!gsg$allOK) expr_t <- expr_t[gsg$goodSamples, gsg$goodGenes]

  sampleTree <- hclust(dist(expr_t), method = "average")
  clust <- cutreeStatic(sampleTree, cutHeight = cut_height, minSize = 10)
  type_code <- substr(rownames(expr_t), 14, 15)
  bar_colors <- cbind(Status = ifelse(clust == 1, "grey85", "black"),
                      Type = ifelse(type_code == "11", "#3498DB", ifelse(type_code == "01", "#E74C3C", "grey60")))
  pdf(P("figures", "Fig_SampleClustering.pdf"), width = 16, height = 7)
  plotDendroAndColors(sampleTree, bar_colors, groupLabels = c("Kept / removed", "Normal / Tumor"),
                      dendroLabels = FALSE, hang = 0.03, addGuide = TRUE, guideHang = 0.05,
                      abHeight = cut_height, abCol = "red",
                      main = paste0("Sample clustering (cut height = ", cut_height, ")"))
  dev.off()
  cat("Removed samples:", sum(clust != 1), "\n")
  expr_clean <- expr_t[clust == 1, ]

  sample_code <- substr(rownames(expr_clean), 14, 15)
  traitData <- data.frame(Normal = as.numeric(sample_code == "11"),
                          Tumor  = as.numeric(sample_code == "01"),
                          row.names = rownames(expr_clean))
  print(colSums(traitData))

  # soft threshold (Fig 1B)
  powers <- c(1:10, seq(12, 20, 2))
  sft <- pickSoftThreshold(expr_clean, powerVector = powers, networkType = "unsigned", verbose = 5)
  pdf(P("figures", "Fig1B_SoftThreshold.pdf"), width = 10, height = 5)
  par(mfrow = c(1, 2))
  plot(sft$fitIndices[, 1], -sign(sft$fitIndices[, 3]) * sft$fitIndices[, 2], type = "n",
       xlab = "Soft Threshold (power)", ylab = "Scale Free Topology Model Fit (R^2)", main = "Scale independence")
  text(sft$fitIndices[, 1], -sign(sft$fitIndices[, 3]) * sft$fitIndices[, 2], labels = powers, col = "red")
  abline(h = 0.8, col = "red")
  plot(sft$fitIndices[, 1], sft$fitIndices[, 5], type = "n",
       xlab = "Soft Threshold (power)", ylab = "Mean Connectivity", main = "Mean connectivity")
  text(sft$fitIndices[, 1], sft$fitIndices[, 5], labels = powers, col = "red")
  dev.off()

  # scale-free check at the chosen power (Fig 1A), RAM-safe
  k <- softConnectivity(expr_clean, power = soft_power, type = "unsigned")
  pdf(P("figures", "Fig1A_ScaleFree.pdf"), width = 10, height = 5)
  par(mfrow = c(1, 2), mar = c(5, 5, 4, 2))
  hist(k, breaks = 30, main = "Histogram of k", xlab = "k", col = "lightgray", border = "gray40")
  scaleFreePlot(k, main = paste0("Check scale free topology (power ", soft_power, ")\n"), no.breaks = 10)
  dev.off()

  cat("Building network (10-40 min)...\n")
  net <- blockwiseModules(expr_clean, power = soft_power, networkType = "unsigned", TOMType = "unsigned",
                          maxBlockSize = max_block_size, minModuleSize = min_module_size,
                          mergeCutHeight = merge_cut, numericLabels = TRUE, pamRespectsDendro = FALSE,
                          saveTOMs = TRUE, saveTOMFileBase = P("rds", "LIHC_TOM"), verbose = 3)
  mergedColors <- labels2colors(net$colors)
  names(mergedColors) <- colnames(expr_clean)
  save(net, expr_clean, traitData, mergedColors, soft_power, file = net_file)
}

cor_T <- cor(MEs, traitData$Tumor, use = "p");               p_T <- corPvalueStudent(cor_T, nrow(expr_clean))

hm_cor <- cbind("Tumor vs Normal" = cor_T[, 1], cor_I)

txt <- matrix(paste0(signif(hm_cor, 2), "\n(", signif(hm_p, 1), ")"), nrow = nrow(hm_cor))

  in_mod <- mergedColors == mc
  mm <- abs(geneModuleMembership[in_mod, paste0("ME", mc)]); gs <- abs(gs_immune[in_mod])

  if (length(mm) < 3) next
  ct <- cor.test(mm, gs)
  plot(mm, gs, type = "n", xlim = range(c(mm, 0.8)), ylim = range(c(gs, gs_cut)),

  mtext(paste0("ImmuneScore(", mc, ")"), side = 3, line = 1.7, font = 2, cex = 0.8)
  mtext(paste0("cor=", round(ct$estimate, 2), ", p=", signif(ct$p.value, 2)), side = 3, line = 0.6, font = 2, cex = 0.7)

# ---- network figure (Fig 2A-C) ---------------------------------------------------
enrich_network <- function(obj, title, file, top_n = 25, sim_cut = 0.1) {
  if (is.null(obj) || nrow(as.data.frame(obj)) < 2) { message("Too few terms: ", title); return(invisible(NULL)) }
  obj2 <- pairwise_termsim(obj)
  res  <- obj2@result %>% dplyr::filter(p.adjust < 0.05) %>% dplyr::arrange(p.adjust)
  res  <- res[seq_len(min(top_n, nrow(res))), ]
  terms <- res$Description
  sim <- obj2@termsim[terms, terms]
  sim[is.na(sim)] <- 0; sim[sim < sim_cut] <- 0; diag(sim) <- 0
  g <- graph_from_adjacency_matrix(sim, mode = "undirected", weighted = TRUE, diag = FALSE)
  if (ecount(g) == 0) { message("No edges above sim_cut for: ", title); return(invisible(NULL)) }

  set.seed(SEED)
  xy <- as.data.frame(layout_with_fr(g)); colnames(xy) <- c("x", "y"); xy$name <- V(g)$name
  nodes <- xy
  nodes$p.adjust <- res$p.adjust[match(nodes$name, res$Description)]
  nodes$size     <- res$Count[match(nodes$name, res$Description)]
  nodes$cluster  <- as.factor(membership(cluster_louvain(g)))

  el <- as.data.frame(as_edgelist(g)); colnames(el) <- c("from", "to")
  edges <- merge(el, xy, by.x = "from", by.y = "name")
  edges <- merge(edges, xy, by.x = "to", by.y = "name", suffixes = c("_from", "_to"))

  labs_df <- nodes %>% dplyr::group_by(cluster) %>%
    dplyr::slice_min(p.adjust, n = 1, with_ties = FALSE) %>% dplyr::ungroup() %>%
    dplyr::select(cluster, label = name)
  cen <- nodes %>% dplyr::group_by(cluster) %>%
    dplyr::summarise(x0 = mean(x), y0 = mean(y),
                     a = max((max(x) - min(x)) / 2 + 0.3, 0.3),
                     b = max((max(y) - min(y)) / 2 + 0.3, 0.3), .groups = "drop")
  labs_df <- dplyr::left_join(labs_df, cen, by = "cluster")

  p <- ggplot() +
    geom_ellipse(data = cen, aes(x0 = x0, y0 = y0, a = a, b = b, angle = 0),
                 fill = NA, colour = "black", linetype = "dashed", linewidth = 0.4) +
    geom_segment(data = edges, aes(x = x_from, y = y_from, xend = x_to, yend = y_to),
                 colour = "grey60", linewidth = 0.3, alpha = 0.7) +
    geom_point(data = nodes, aes(x = x, y = y, size = size, colour = p.adjust), alpha = 0.9) +
    geom_text(data = labs_df, aes(x = x0 + 0.4, y = y0 + 0.4, label = str_wrap(label, 25)),
              size = 3, fontface = "bold", hjust = 0) +
    scale_size_continuous(name = "Pathway size", range = c(2, 8)) +
    scale_colour_gradient(name = "p.adjust", low = "#D73027", high = "#4575B4", trans = "reverse") +
    ggtitle(title) + theme_void() +
    theme(plot.title = element_text(hjust = 0.5, size = 14, face = "bold"))
  ggsave(file, plot = p, width = 12, height = 10, dpi = 300)
  invisible(p)
}

# 07_hub_genes_immune.R
# Hub genes = ML feature genes  ∩  IMMUNE-RESTRICTED scRNA DEGs
#
# Logic (differs from the old script, which used DEGs from ALL clusters):
#   1. DEGs are taken from immune clusters only (T, NK, B, macrophage, monocyte,
#      neutrophil) and genes that are strong markers of non-immune clusters
#      (hepatocytes, endothelial, etc.) are removed.
#   2. Each candidate must be directly verified as immune-enriched in the raw
#      scRNA data (mean ratio immune/non-immune and % cells expressing).
#   3. Candidates come from the ML pool in tiers (strict -> relaxed) so you
#      always get a transparent, reportable gene list.
# Input : results/GSE299340_all_markers_deg.csv, HCC_ML_gene_overlap.csv,
#         GSE299340_cluster_celltypes.csv, HCC_coexpressed_genes.csv,
#         rds/GSE299340_annotated.rds
# Output: results/HCC_hub_genes.csv  (same name as before: downstream scripts work)
#         results/hub_gene_immune_specificity.csv
# =============================================================================
source("D:/Bioinfo/scripts/config.R")
library(dplyr)
library(Seurat)

# ---- settings ----------------------------------------------------------------
immune_types   <- c("T_cells", "NK_cell", "B_cell", "Macrophage", "Monocyte", "Neutrophils")
min_ratio      <- 1.5    # mean expression immune / non-immune
min_pct_immune <- 5      # % of immune cells expressing the gene
n_hub_max      <- 30     # cap on hub genes

all_markers_deg <- read.csv(P("results", "GSE299340_all_markers_deg.csv"))
gene_counts     <- read.csv(P("results", "HCC_ML_gene_overlap.csv"))
cluster_anno    <- read.csv(P("results", "GSE299340_cluster_celltypes.csv"))
coexp           <- read.csv(P("results", "HCC_coexpressed_genes.csv"))   # Gene, Module, GS_Immune, GS_Tumor

# ---- 1. immune-restricted DEGs -----------------------------------------------
cluster_anno$cluster <- as.character(cluster_anno$cluster)
all_markers_deg$cluster <- as.character(all_markers_deg$cluster)
imm_cl <- cluster_anno$cluster[cluster_anno$cell_type %in% immune_types]
cat("Immune clusters:", paste(imm_cl, collapse = ", "), "\n")
stopifnot(length(imm_cl) > 0)

imm_up <- all_markers_deg %>%
  filter(cluster %in% imm_cl, p_val_adj < 0.05, avg_log2FC > 0.5) %>% pull(gene) %>% unique()
other_up <- all_markers_deg %>%
  filter(!cluster %in% imm_cl, p_val_adj < 0.05, avg_log2FC > 1) %>% pull(gene) %>% unique()
imm_deg <- setdiff(imm_up, other_up)
cat("Immune up-regulated DEGs:", length(imm_up), "| after removing non-immune markers:", length(imm_deg), "\n")

# ---- 2. verify immune enrichment in the raw scRNA data -----------------------
combined <- readRDS(P("rds", "GSE299340_annotated.rds"))
DefaultAssay(combined) <- "RNA"
is_imm <- combined$cell_type %in% immune_types

cand_all <- intersect(intersect(imm_deg, coexp$Gene), rownames(combined))
cat("Immune DEGs inside the WGCNA modules:", length(cand_all), "\n")
if (length(cand_all) < 3)
  stop("Fewer than 3 immune genes inside the purple/red modules. The tumor modules contain ",
        "almost no immune genes: lower min_ratio, or raise n_modules / lower imm_cor_cutoff in script 02 to include more modules. ",
        "Immune modules are chosen by correlation with ImmuneScore.")

ex <- FetchData(combined, vars = cand_all)                 # cells x genes (log-normalised)
lin <- expm1(as.matrix(ex))
spec <- data.frame(
  gene          = cand_all,
  mean_immune   = colMeans(lin[is_imm, , drop = FALSE]),
  mean_other    = colMeans(lin[!is_imm, , drop = FALSE]),
  pct_immune    = colMeans(lin[is_imm, , drop = FALSE] > 0) * 100,
  pct_other     = colMeans(lin[!is_imm, , drop = FALSE] > 0) * 100)
spec$ratio <- spec$mean_immune / (spec$mean_other + 1e-3)
spec$immune_specific <- spec$ratio >= min_ratio & spec$pct_immune >= min_pct_immune
spec$ML_count <- gene_counts$Count[match(spec$gene, gene_counts$Gene)]
spec$ML_count[is.na(spec$ML_count)] <- 0
spec$abs_GS <- abs(coexp$GS_Immune[match(spec$gene, coexp$Gene)])
write.csv(spec[order(-spec$ratio), ], P("results", "hub_gene_immune_specificity.csv"), row.names = FALSE)
rm(combined, ex, lin); gc()

# ---- 3. tiered selection ------------------------------------------------------
tiers <- list(
  T1_selected_by_2plus_ML = spec$ML_count >= 2,
  T2_selected_by_any_ML   = spec$ML_count >= 1,
  T3_module_genes_only    = rep(TRUE, nrow(spec)))
hub <- NULL
for (t in names(tiers)) {
  s <- spec[tiers[[t]] & spec$immune_specific, ]
  cat(t, ":", nrow(s), "immune-specific genes\n")
  if (nrow(s) >= 3) {
    s <- s[order(-s$ML_count, -s$abs_GS), ]
    hub <- head(s, n_hub_max); hub$tier <- t
    break
  }
}
if (is.null(hub)) stop("No tier gives >= 3 immune-specific genes. Relax min_ratio / min_pct_immune or redo script 02.")

cat("\nTier used:", hub$tier[1], "| hub genes:", nrow(hub), "\n")
print(hub[, c("gene", "ML_count", "ratio", "pct_immune", "pct_other", "tier")])
write.csv(data.frame(HubGene = hub$gene, Tier = hub$tier, ML_count = hub$ML_count,
                     ImmuneRatio = round(hub$ratio, 2)),
          P("results", "HCC_hub_genes.csv"), row.names = FALSE)
cat("Done.\n")

# MCP-counter + CIBERSORTx on bulk TCGA-LIHC; hub gene vs immune cell correlations
#
# Changes vs. old script:
#   * correlations are computed WITHIN TUMOR samples only (using Tumor+Normal
#     together makes every tumor gene look "immune-correlated")

# Spearman correlation of every hub gene with every cell score, tumor samples only
cor_table <- function(cell_df, hub_df, cells, ids, source) {
  m <- inner_join(cell_df[cell_df$Mixture %in% ids, c("Mixture", cells)],
                  hub_df[hub_df$Mixture %in% ids, ], by = "Mixture")
  cells <- cells[sapply(m[, cells, drop = FALSE], function(x) sd(x, na.rm = TRUE) > 0)]
  genes <- setdiff(colnames(hub_df), "Mixture")
  out <- do.call(rbind, lapply(genes, function(g) do.call(rbind, lapply(cells, function(ce) {
    ct <- suppressWarnings(cor.test(m[[g]], m[[ce]], method = "spearman", exact = FALSE))
    data.frame(gene = g, cell = ce, rho = unname(ct$estimate), p = ct$p.value, source = source)
  }))))
  out$fdr <- p.adjust(out$p, "BH")
  out
}

fan_plot <- function(gene, cell_df, hub_df, cells, ids, file) {
  m <- inner_join(cell_df[cell_df$Mixture %in% ids, c("Mixture", cells)],
                  hub_df[hub_df$Mixture %in% ids, c("Mixture", gene)], by = "Mixture")
  cells <- cells[sapply(m[, cells], function(x) sd(x, na.rm = TRUE) > 0)]
  cell_cor <- correlate(m[, cells], method = "spearman")
  mt <- mantel_test(m[, gene, drop = FALSE], m[, cells], spec_select = setNames(list(1), gene)) %>%
    mutate(rd = cut(r, breaks = c(-Inf, 0.2, 0.4, Inf), labels = c("< 0.2", "0.2 - 0.4", ">= 0.4")),
           pd = cut(p, breaks = c(-Inf, 0.01, 0.05, Inf), labels = c("< 0.01", "0.01 - 0.05", ">= 0.05")))
  p <- qcorrplot(cell_cor, grid_col = "grey70", grid_size = 0.2) +
    geom_square() +
    geom_couple(aes(colour = pd, size = rd), data = mt, curvature = 0.1) +
    scale_fill_gradientn(colours = c("#4575B4", "white", "#D73027"), limits = c(-1, 1), name = "Spearman's R") +
    scale_size_manual(values = c(0.4, 0.8, 1.6), name = "Mantel r") +
    scale_colour_manual(values = c("#D73027", "#FDAE61", "grey80"), name = "P-value") +
    ggtitle(paste0(gene, " (tumor samples)")) + theme(plot.title = element_text(face = "bold", hjust = 0))
  ggsave(file, p, width = 9, height = 8, dpi = 300)
}

boxplot_groups <- function(long, title, ylab, file) {
  p <- ggplot(long, aes(Cell_Type, Value, fill = group)) +
    geom_boxplot(outlier.size = 0.3, width = 0.6) +
    stat_compare_means(aes(group = group), method = "wilcox.test", label = "p.signif", hide.ns = TRUE) +
    scale_fill_manual(values = c(Tumor = "#E74C3C", Normal = "#3498DB")) +
    labs(title = title, x = NULL, y = ylab, fill = "Group") + theme_bw() +
    theme(plot.title = element_text(hjust = 0.5, face = "bold"),
          axis.text.x = element_text(angle = 45, hjust = 1), panel.grid = element_blank())
  ggsave(file, p, width = 13, height = 6, dpi = 300)
}
