# Check the candidate cutoffs
candidate_heights <- c(200, 250, 300, 350, 400, 500)

cutoff_results <- do.call(
  rbind,
  lapply(candidate_heights, function(h) {
    
    ct <- cutreeStatic(
      sampleTree,
      cutHeight = h,
      minSize = 10
    )
    
    tab <- table(ct)
    
    data.frame(
      Cutoff = h,
      Total_samples = length(ct),
      Main_cluster = sum(ct == 1),
      Flagged_or_other = sum(ct != 1),
      Number_of_clusters = length(tab),
      Cluster_sizes = paste(
        names(tab), as.integer(tab),
        sep = ":",
        collapse = "; "
      )
    )
  })
)

for (mc in sig_module_colors) {
  in_mod <- mergedColors == mc
  mm <- abs(geneModuleMembership[in_mod, paste0("ME", mc)])
  gs <- abs(geneTraitSignificance[in_mod, "GS.Tumor"])
  ok <- complete.cases(mm, gs)
  if (sum(ok) < 3) next
  mm <- mm[ok]; gs <- gs[ok]
  ct <- cor.test(mm, gs)
  
  plot(mm, gs, type = "n",
       xlim = range(c(mm, mm_cut)), ylim = range(c(gs, gs_cut)),
       xlab = "Module membership", ylab = "Gene significance",
       main = "", bty = "o")
  points(mm, gs, pch = 1, cex = dot_cex, lwd = dot_lwd,
         col = if (is.null(dot_col)) mc else dot_col)
  abline(v = mm_cut, h = gs_cut, lty = 2, lwd = 1, col = "black")
  mtext(paste0("Tumor(", mc, ")"), side = 3, line = 1.7, font = 2, cex = 0.8)
  mtext(paste0("cor=", round(ct$estimate, 2), ", p=", signif(ct$p.value, 2)),
        side = 3, line = 0.6, font = 2, cex = 0.7)
}

# ---- Network figure (Fig 2A-C) ---------------------------------------------
enrich_network <- function(obj, title, file, top_n = 25, sim_cut = 0.1) {
  obj2 <- pairwise_termsim(obj)
  res  <- obj2@result %>% filter(p.adjust < 0.05) %>% arrange(p.adjust)
  res  <- res[seq_len(min(top_n, nrow(res))), ]
  terms <- res$Description
  sim <- obj2@termsim[terms, terms]
  sim[is.na(sim)] <- 0
  sim[sim < sim_cut] <- 0
  diag(sim) <- 0
  g <- graph_from_adjacency_matrix(sim, mode = "undirected", weighted = TRUE, diag = FALSE)
  
  set.seed(SEED)
  xy <- as.data.frame(layout_with_fr(g)); colnames(xy) <- c("x", "y"); xy$name <- V(g)$name
  nodes <- xy
  nodes$p.adjust <- res$p.adjust[match(nodes$name, res$Description)]
  nodes$size     <- res$Count[match(nodes$name, res$Description)]
  nodes$cluster  <- as.factor(membership(cluster_louvain(g)))
  
  el <- as.data.frame(as_edgelist(g)); colnames(el) <- c("from", "to")
  edges <- merge(el, xy, by.x = "from", by.y = "name")
  edges <- merge(edges, xy, by.x = "to", by.y = "name", suffixes = c("_from", "_to"))
  
  labs_df <- nodes %>% group_by(cluster) %>% slice_min(p.adjust, n = 1, with_ties = FALSE) %>%
    ungroup() %>% select(cluster, label = name)
  cen <- nodes %>% group_by(cluster) %>%
    summarise(x0 = mean(x), y0 = mean(y),
              a = max((max(x) - min(x)) / 2 + 0.3, 0.3),
              b = max((max(y) - min(y)) / 2 + 0.3, 0.3), .groups = "drop")
  labs_df <- left_join(labs_df, cen, by = "cluster")
  
  p <- ggplot() +
    geom_ellipse(data = cen, aes(x0 = x0, y0 = y0, a = a, b = b, angle = 0),
                 fill = NA, colour = "black", linetype = "dashed", linewidth = 0.4) +
    geom_segment(data = edges, aes(x = x_from, y = y_from, xend = x_to, yend = y_to),
                 colour = "grey60", linewidth = 0.3, alpha = 0.7) +
    geom_point(data = nodes, aes(x = x, y = y, size = size, colour = p.adjust), alpha = 0.9) +
    geom_text(data = labs_df, aes(x = x0 + 0.4, y = y0 + 0.4, label = str_wrap(label, 25)),
              size = 3, fontface = "bold", hjust = 0) +
    scale_size_continuous(name = "Pathway size", range = c(2, 8)) +
    scale_colour_gradient(name = "p.adjust", low = "#D73027", high = "#4575B4", trans = "reverse") +
    ggtitle(title) + theme_void() +
    theme(plot.title = element_text(hjust = 0.5, size = 14, face = "bold"))
  ggsave(file, plot = p, width = 12, height = 10, dpi = 300)
  invisible(p)
}

# Function to plot normalized confusion matrix
plot_cm <- function(pred_prob, true_labels, title, filename) {
  pred_class  <- factor(ifelse(pred_prob > 0.5, "Tumor", "Normal"), levels=c("Tumor","Normal"))
  true_class  <- factor(ifelse(true_labels == 1, "Tumor", "Normal"), levels=c("Tumor","Normal"))
  cm          <- confusionMatrix(pred_class, true_class)
  cm_norm     <- cm$table / rowSums(cm$table)
  cm_df       <- as.data.frame(cm_norm)
  colnames(cm_df) <- c("Predicted", "Actual", "Proportion")
  p <- ggplot(cm_df, aes(x=Predicted, y=Actual, fill=Proportion)) +
    geom_tile(color="white", linewidth=2) +
    geom_text(aes(label=sprintf("%.2f", Proportion)), size=12, fontface="bold", color="white") +
    scale_fill_viridis(option="viridis", limits=c(0,1)) +
    labs(title="Normalized confusion matrix", x="Predicted", y="True label") +
    theme_bw()
  ggsave(filename, plot=p, width=6, height=5, dpi=300)
  return(p)
}
